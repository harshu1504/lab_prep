:: 1. Build Maven project
mvn clean package

:: 2. Build Docker image
docker build -t foodimage123 .

:: 3. Login to Docker Hub
docker login

:: 4. Tag image
docker tag foodimage123 dockerhubusername/foodimage123:latest

:: 5. Push image
docker push dockerhubusername/foodimage123:latest

:: ON ANOTHER SYSTEM

:: 6. Login
docker login

:: 7. Pull image
docker pull dockerhubusername/foodimage123:latest

:: 8. Run application
docker run -d -p 8081:8080 --name foodcontainer dockerhubusername/foodimage123:latest

:: 9. Verify
docker ps

:: 10. Open browser
:: http://localhost:8081/application/






:: Start Minikube if using Minikube
minikube start

:: Create NGINX deployment
kubectl create deployment mynzinx --image=nginx

:: Check deployment and pods
kubectl get deployments
kubectl get pods

:: Scale to 3 replicas
kubectl scale deployment mynzinx --replicas=3

:: Scale to 5 replicas
kubectl scale deployment mynzinx --replicas=5

:: Scale down to 2 replicas
kubectl scale deployment mynzinx --replicas=2

:: Expose deployment
kubectl expose deployment mynzinx --type=NodePort --port=80

:: Check services
kubectl get services

:: Access using port forwarding
kubectl port-forward service/mynzinx 8080:80

:: Delete resources
kubectl delete service mynzinx
kubectl delete deployment mynzinx
